#!/usr/bin/env node
// Rename this file to import-vocabulary.mjs before running.
// Usage: npm install pg && DATABASE_URL=... node import-vocabulary.mjs
// Server-side only. Idempotent source upsert; imported app entries remain drafts.
import pg from "pg";
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL, ssl: process.env.PGSSLMODE === "disable" ? false : { rejectUnauthorized: false } });
const base = "https://raw.githubusercontent.com/Maximax67/Words-CEFR-Dataset/main/csv/";
const usUrl = "https://raw.githubusercontent.com/open-dict-data/ipa-dict/master/data/en_US.txt";
const ukUrl = "https://raw.githubusercontent.com/open-dict-data/ipa-dict/master/data/en_UK.txt";
if (!process.env.DATABASE_URL) throw new Error("DATABASE_URL is required.");

function csvLine(line) {
  const out=[]; let value="", quoted=false;
  for(let i=0;i<line.length;i++){const c=line[i];if(c==='"'){if(quoted&&line[i+1]==='"'){value+='"';i++;}else quoted=!quoted;}else if(c===","&&!quoted){out.push(value);value="";}else value+=c;}
  out.push(value);return out;
}
function level(n){if(!Number.isFinite(n)||n<=0)return null;if(n<1.5)return "A1";if(n<2.5)return "A2";if(n<3.5)return "B1";if(n<4.5)return "B2";if(n<5.5)return "C1";return "C2";}
async function get(url){const r=await fetch(url);if(!r.ok)throw new Error("Fetch "+r.status+" "+url);return r.text();}
function ipaMap(text){const m=new Map();for(const line of text.split(/\r?\n/)){const i=line.indexOf("\t");if(i<0)continue;const w=line.slice(0,i).trim().normalize("NFKC").toLowerCase(),v=line.slice(i+1).trim();if(w&&v&&!m.has(w))m.set(w,v);}return m;}

async function main(){
 const [wc,pc,tc,usText,ukText]=await Promise.all([get(base+"words.csv"),get(base+"word_pos.csv"),get(base+"pos_tags.csv"),get(usUrl),get(ukUrl)]);
 const words=new Map(),tags=new Map();
 for(const line of wc.split(/\r?\n/).slice(1)){if(!line)continue;const r=csvLine(line);if(r[0]&&r[1])words.set(r[0],r[1]);}
 for(const line of tc.split(/\r?\n/).slice(1)){if(!line)continue;const r=csvLine(line);if(r[0])tags.set(r[0],r[1]||"");}
 const us=ipaMap(usText),uk=ipaMap(ukText),rows=pc.split(/\r?\n/).slice(1).filter(Boolean);
 let staged=0;
 for(let offset=0;offset<rows.length;offset+=2500){
  const batch=[];
  for(const line of rows.slice(offset,offset+2500)){
   const r=csvLine(line),term=words.get(r[1]);if(!term)continue;
   const score=r[5]===""?null:Number(r[5]),norm=term.normalize("NFKC").trim().toLowerCase();
   batch.push({source_key:"words-cefr-dataset",source_record_id:r[0],term,normalized_term:norm,part_of_speech:tags.get(r[2])||"",source_level_score:Number.isFinite(score)?score:null,cefr_level:level(score),frequency_count:/^\d+$/.test(r[4]||"")?Number(r[4]):null,ipa_us:us.get(norm)||null,ipa_uk:uk.get(norm)||null,source_url:"https://github.com/Maximax67/Words-CEFR-Dataset",license:"MIT repository license; upstream acknowledgements retained",raw_metadata:{word_id:r[1],pos_tag_id:r[2],lemma_word_id:r[3]||null,source_level:r[5]||null}});
  }
  if(!batch.length)continue;
  const sql=[
   "INSERT INTO vocabulary_source_records (source_key,source_record_id,term,normalized_term,part_of_speech,source_level_score,cefr_level,frequency_count,ipa_us,ipa_uk,source_url,license,raw_metadata,ipa_us_source_url,ipa_us_license,ipa_uk_source_url,ipa_uk_license)",
   "SELECT x.source_key,x.source_record_id,x.term,x.normalized_term,x.part_of_speech,x.source_level_score,x.cefr_level,x.frequency_count,x.ipa_us,x.ipa_uk,x.source_url,x.license,x.raw_metadata,$2,$3,$4,$5",
   "FROM jsonb_to_recordset($1::jsonb) AS x(source_key text,source_record_id text,term text,normalized_term text,part_of_speech text,source_level_score numeric,cefr_level text,frequency_count bigint,ipa_us text,ipa_uk text,source_url text,license text,raw_metadata jsonb)",
   "ON CONFLICT (source_key,source_record_id) DO UPDATE SET term=EXCLUDED.term,normalized_term=EXCLUDED.normalized_term,part_of_speech=EXCLUDED.part_of_speech,source_level_score=EXCLUDED.source_level_score,cefr_level=EXCLUDED.cefr_level,frequency_count=EXCLUDED.frequency_count,ipa_us=EXCLUDED.ipa_us,ipa_uk=EXCLUDED.ipa_uk,raw_metadata=EXCLUDED.raw_metadata,ipa_us_source_url=EXCLUDED.ipa_us_source_url,ipa_us_license=EXCLUDED.ipa_us_license,ipa_uk_source_url=EXCLUDED.ipa_uk_source_url,ipa_uk_license=EXCLUDED.ipa_uk_license"
  ].join(" ");
  await pool.query(sql,[JSON.stringify(batch),"https://github.com/open-dict-data/ipa-dict","MIT for en_US per repository credits","https://github.com/open-dict-data/ipa-dict","GPL-3.0 for en_UK per repository credits (ipacards-derived)"]);
  staged+=batch.length;if(offset%25000===0)console.log("Staged "+Math.min(offset+batch.length,rows.length)+" / "+rows.length);
 }
 await pool.query(
  "INSERT INTO vocabulary_import_sources (source_key,source_name,source_url,license,attribution,version_label,notes,record_count,imported_at) VALUES " +
  "('words-cefr-dataset','Words-CEFR-Dataset','https://github.com/Maximax67/Words-CEFR-Dataset','MIT repository license; upstream acknowledgements retained','Maxim Belikov / Maximax67; see README for upstream sources','main snapshot','Algorithmic CEFR estimates; preserve raw score and source row ID'," + rows.length + ",now())," +
  "('ipa-dict-en-us','ipa-dict English US','https://github.com/open-dict-data/ipa-dict','MIT for en_US per repository credits','open-dict-data/ipa-dict; modified cmudict-ipa-derived','master snapshot','Broadly phonemic US pronunciation'," + us.size + ",now())," +
  "('ipa-dict-en-uk','ipa-dict English UK','https://github.com/open-dict-data/ipa-dict','GPL-3.0 per repository credits (ipacards-derived)','open-dict-data/ipa-dict; ipacards upstream','master snapshot','Track license separately; assess distribution obligations'," + uk.size + ",now()) " +
  "ON CONFLICT (source_key) DO UPDATE SET record_count=EXCLUDED.record_count,imported_at=EXCLUDED.imported_at"
 );
 const promote=[
 "WITH ranked AS (SELECT s.*,ROW_NUMBER() OVER (PARTITION BY normalized_term ORDER BY CASE cefr_level WHEN 'A1' THEN 1 WHEN 'A2' THEN 2 WHEN 'B1' THEN 3 WHEN 'B2' THEN 4 ELSE 9 END,COALESCE(frequency_count,0) DESC,source_record_id) rn",
 "FROM vocabulary_source_records s WHERE source_key='words-cefr-dataset' AND cefr_level IN ('A1','A2','B1','B2') AND normalized_term ~ '^[a-z]+$' AND part_of_speech NOT IN ('NNP','NNPS'))",
 "INSERT INTO vocabulary_entries (term,normalized_term,entry_type,ipa,part_of_speech,definition_en,meaning_vi,cefr_level,example,source_name,source_url,source_status)",
 "SELECT term,normalized_term,'word',COALESCE(NULLIF(ipa_us,''),NULLIF(ipa_uk,''),''),CASE WHEN part_of_speech IN ('NN','NNS') THEN 'noun' WHEN part_of_speech IN ('VB','VBD','VBG','VBN','VBP','VBZ') THEN 'verb' WHEN part_of_speech IN ('JJ','JJR','JJS') THEN 'adjective' WHEN part_of_speech IN ('RB','RBR','RBS') THEN 'adverb' WHEN part_of_speech IN ('IN','TO') THEN 'preposition' WHEN part_of_speech IN ('PRP','PRP$') THEN 'pronoun' ELSE 'other' END,'',NULL,cefr_level,'','Words-CEFR-Dataset (draft import)','https://github.com/Maximax67/Words-CEFR-Dataset','draft' FROM ranked WHERE rn=1 ON CONFLICT (normalized_term) DO NOTHING"
 ].join(" ");
 const result=await pool.query(promote);
 console.log(JSON.stringify({sourceRows:rows.length,stagedRows:staged,wordForms:words.size,ipaUS:us.size,ipaUK:uk.size,newDraftEntries:result.rowCount}));
 await pool.end();
}
main().catch(async e=>{console.error(e);await pool.end();process.exitCode=1;});
